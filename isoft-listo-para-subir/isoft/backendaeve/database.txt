-- Tabla: cargo
CREATE TABLE cargo (
    id SERIAL PRIMARY KEY,

CREATE TABLE rol (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    descripcion TEXT
);

-- Tabla: contacto
CREATE TABLE contacto (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(100),
    apellido VARCHAR(100),
    cargo VARCHAR(100),
    email VARCHAR(150) CONSTRAINT check_email_contacto CHECK (
        email ~* '^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.(com|cl)$'
    ),
    telefono VARCHAR(20),
    principal BOOLEAN DEFAULT FALSE
);

-- Tabla: proveedor
CREATE TABLE proveedor (
    id SERIAL PRIMARY KEY,
    rut VARCHAR(12) UNIQUE,
    nombre VARCHAR(200) NOT NULL UNIQUE,
    direccion VARCHAR(255),
    telefono VARCHAR(20),
    web VARCHAR(255),
    email VARCHAR(150) CONSTRAINT check_email_proveedor CHECK (
        email ~* '^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.(com|cl)$'
    ),
    descripcion TEXT
);

-- Tabla: tipo_cliente
CREATE TABLE tipo_cliente (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    descripcion TEXT
);

-- Tabla: cliente
CREATE TABLE cliente (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(100),
    rut VARCHAR(12) NOT NULL UNIQUE CONSTRAINT check_rut_format CHECK (
        rut ~* '^[0-9]{1,2}\.[0-9]{3}\.[0-9]{3}-[0-9Kk]$'
    ),
    razon_social VARCHAR(200),
    apellidos VARCHAR(100),
    email VARCHAR(150) CONSTRAINT check_email_cliente CHECK (
        email ~* '^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.(com|cl)$'
    ),
    comuna VARCHAR(100),
    direccion VARCHAR(255),
    ciudad VARCHAR(100),
    activo BOOLEAN DEFAULT TRUE,
    fecha_registro DATE DEFAULT CURRENT_DATE,
    tipo_cliente_id INTEGER,
    CONSTRAINT fk_cliente_tipo FOREIGN KEY (tipo_cliente_id) 
        REFERENCES tipo_cliente(id) ON DELETE SET NULL
);

-- Tabla: contacto_empresa (relación muchos a muchos)
CREATE TABLE contacto_empresa (
    id SERIAL PRIMARY KEY,
    id_contacto INTEGER NOT NULL,
    id_empresa INTEGER NOT NULL,
    CONSTRAINT fk_contacto FOREIGN KEY (id_contacto) 
        REFERENCES contacto(id) ON DELETE CASCADE,
    CONSTRAINT fk_cliente FOREIGN KEY (id_empresa) 
        REFERENCES cliente(id) ON DELETE CASCADE,
    CONSTRAINT unique_contacto_empresa UNIQUE (id_contacto, id_empresa)
);

-- Tabla: tipo_producto
CREATE TABLE tipo_producto (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    descripcion TEXT
);

-- Tabla: producto
CREATE TABLE producto (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(200) NOT NULL,
    imagen BYTEA,
    codigo_ska VARCHAR(50),
    unidades INTEGER DEFAULT 0,
    descripcion TEXT,
    tipo_producto_id INTEGER,
    CONSTRAINT fk_producto_tipo FOREIGN KEY (tipo_producto_id) 
        REFERENCES tipo_producto(id) ON DELETE SET NULL
);

-- Tabla: materiales_producto
CREATE TABLE materiales_producto (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(200),
    unidades INTEGER DEFAULT 0,
    descripcion TEXT,
    proveedor_id INTEGER,
    tipo_producto_id INTEGER,
    CONSTRAINT fk_material_proveedor FOREIGN KEY (proveedor_id) 
        REFERENCES proveedor(id) ON DELETE SET NULL,
    CONSTRAINT fk_material_tipo FOREIGN KEY (tipo_producto_id) 
        REFERENCES tipo_producto(id) ON DELETE SET NULL
);

-- Tabla: usuario
CREATE TABLE usuario (
    id SERIAL PRIMARY KEY,
    nombre VARCHAR(100),
    apellidos VARCHAR(100),
    email VARCHAR(150) UNIQUE NOT NULL CONSTRAINT check_email_usuario CHECK (
        email ~* '^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.(com|cl)$'
    ),
    rut VARCHAR(12) UNIQUE CONSTRAINT check_rut_usuario CHECK (
        rut ~* '^[0-9]{1,2}\.[0-9]{3}\.[0-9]{3}-[0-9Kk]$'
    ),
    telefono VARCHAR(20),
    direccion VARCHAR(255),
    cargo_id INTEGER,
    rol_id INTEGER,
    CONSTRAINT fk_usuario_cargo FOREIGN KEY (cargo_id) 
        REFERENCES cargo(id) ON DELETE SET NULL,
    CONSTRAINT fk_usuario_rol FOREIGN KEY (rol_id) 
        REFERENCES rol(id) ON DELETE SET NULL
);

-- Tabla: contraseña (con encriptación)
CREATE TABLE contrasena (
    id SERIAL PRIMARY KEY,
    contrasena_temporal VARCHAR(255),
    contrasena VARCHAR(255) NOT NULL,
    fecha_ultima_contrasena TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    usuario_id INTEGER NOT NULL,
    CONSTRAINT fk_contrasena_usuario FOREIGN KEY (usuario_id) 
        REFERENCES usuario(id) ON DELETE CASCADE
);

-- Función para encriptar contraseñas antes de insertar
CREATE OR REPLACE FUNCTION encriptar_contrasena()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.contrasena IS NOT NULL THEN
        NEW.contrasena := crypt(NEW.contrasena, gen_salt('bf', 8));
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- Trigger para encriptar automáticamente
CREATE TRIGGER trigger_encriptar_contrasena
    BEFORE INSERT OR UPDATE ON contrasena
    FOR EACH ROW
    EXECUTE FUNCTION encriptar_contrasena();

-- Función para validar contraseña
CREATE OR REPLACE FUNCTION validar_contrasena(usuario_id_param INTEGER, contrasena_param VARCHAR)
RETURNS BOOLEAN AS $$
DECLARE
    contrasena_hash VARCHAR;
BEGIN
    SELECT contrasena INTO contrasena_hash 
    FROM contrasena 
    WHERE usuario_id = usuario_id_param 
    ORDER BY fecha_ultima_contrasena DESC 
    LIMIT 1;
    
    RETURN (contrasena_hash = crypt(contrasena_param, contrasena_hash));
END;
$$ LANGUAGE plpgsql;
